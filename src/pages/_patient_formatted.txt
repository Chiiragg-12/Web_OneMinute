import { useEffect, useState, type FormEvent } from 'react';
import { Link, useNavigate, useParams } from 'react-router-dom';
import { api } from '../api/client';
import { useAuth } from '../auth/AuthContext';
import type { Clinic, Doctor, PublicDoctor } from '../types/api';
import { ApiErrorState, Button, EmptyState, Field, LoadingState, SuccessMessage } from '../components/ui';

export function ProfilePage() {
  const { state, updatePatient } = useAuth(); if (state.status !== 'authenticated' || !state.patient) return null; const patient = state.patient;
  const [form, setForm] = useState({ fullName: patient.fullName, email: patient.email ?? '', phone: patient.phone ?? '', dob: patient.dob?.slice(0, 10) ?? '', gender: patient.gender ?? '' }); const [error, setError] = useState<unknown>(null); const [saved, setSaved] = useState(false); const [busy, setBusy] = useState(false);
  const submit = async (event: FormEvent) => { event.preventDefault(); setError(null); setSaved(false); setBusy(true); try { await updatePatient({ ...form, email: form.email || undefined, phone: form.phone || undefined, dob: form.dob || undefined, gender: form.gender || undefined }); setSaved(true); } catch (nextError) { setError(nextError); } finally { setBusy(false); } };
  return <div>
<div className="eyebrow">Your details</div>
<h1 className="page-title">Profile & Health ID</h1>
<p className="page-lead">Your Health ID is permanent. Contact details can be updated here.</p>
<div className="mt-8 grid gap-6 lg:grid-cols-[0.65fr_1.35fr]">
<section className="health-card bg-teal-950 text-white">
<p className="label text-teal-200">Permanent Health ID</p>
<p className="mt-5 break-all font-display text-3xl font-semibold">{patient.healthId}</p>
<p className="mt-5 text-sm leading-6 text-teal-100/70">This identifier belongs to your patient record and cannot be edited.</p>
</section>
<section className="surface">
<div className="mb-6 flex items-center gap-3">
<span className="icon-tile">
<UserRound size={18} />
</span>
<div>
<h2 className="font-display text-xl font-semibold">Personal information</h2>
<p className="text-sm text-slate-500">Only your own patient profile is shown.</p>
</div>
</div>{error ? <ApiErrorState error={error} /> : null}{saved && <SuccessMessage>Profile updated.</SuccessMessage>}<form className="mt-6 grid gap-5 sm:grid-cols-2" onSubmit={submit}>
<div className="sm:col-span-2">
<Field label="Full name" value={form.fullName} onChange={(event) => setForm({ ...form, fullName: event.target.value })} required />
</div>
<Field label="Email" type="email" value={form.email} onChange={(event) => setForm({ ...form, email: event.target.value })} />
<Field label="Phone" type="tel" value={form.phone} onChange={(event) => setForm({ ...form, phone: event.target.value })} />
<Field label="Date of birth" type="date" value={form.dob} onChange={(event) => setForm({ ...form, dob: event.target.value })} />
<Field label="Gender" value={form.gender} onChange={(event) => setForm({ ...form, gender: event.target.value })} />
<div className="sm:col-span-2">
<Button disabled={busy}>{busy ? 'Saving…' : 'Save changes'}</Button>
</div>
</form>
</section>
</div>
</div>;
}

export function ClinicsPage() {
  const [clinics, setClinics] = useState<Clinic[]>([]); const [error, setError] = useState<unknown>(null); const [loading, setLoading] = useState(true);
  const load = () => { setLoading(true); void api.listClinics().then((result) => setClinics(result.clinics)).catch(setError).finally(() => setLoading(false)); }; useEffect(load, []);
  return <div>
<div className="eyebrow">Find care</div>
<h1 className="page-title">Choose a clinic</h1>
<p className="page-lead">Start with an active clinic, then explore its available doctors.</p>{loading ? <LoadingState label="Finding active clinics" /> : error ? <div className="mt-8">
<ApiErrorState error={error} onRetry={load} />
</div> : clinics.length === 0 ? <div className="mt-8">
<EmptyState title="No clinics listed yet" message="There are no active clinics available right now." />
</div> : <div className="mt-8 grid gap-4 md:grid-cols-2">{clinics.map((clinic) => <Link className="clinic-card" to={`/app/clinics/${clinic.id}`} key={clinic.id}>
<span className="icon-tile">
<MapPin size={19} />
</span>
<span className="min-w-0">
<strong>{clinic.name}</strong>
<span>{clinic.address || 'Address available at the clinic'}</span>
<small>{clinic.timezone}</small>
</span>
<ArrowRight className="ml-auto shrink-0 text-slate-400" size={18} />
</Link>)}</div>}</div>;
}

export function ClinicPage() {
  const { clinicId } = useParams(); const [clinic, setClinic] = useState<Clinic | null>(null); const [doctors, setDoctors] = useState<Doctor[]>([]); const [error, setError] = useState<unknown>(null); const [loading, setLoading] = useState(true);
  useEffect(() => { if (!clinicId) return; void Promise.all([api.getClinic(clinicId), api.listClinicDoctors(clinicId)]).then(([clinicResult, doctorsResult]) => { setClinic(clinicResult.clinic); setDoctors(doctorsResult.doctors); }).catch(setError).finally(() => setLoading(false)); }, [clinicId]);
  if (loading) return <LoadingState label="Loading clinic" />; if (error) return <ApiErrorState error={error} />; if (!clinic) return <EmptyState title="Clinic not found" message="This clinic is no longer available." />;
  return <div>
<Link className="back-link" to="/app/clinics">
<ArrowLeft size={16} />All clinics</Link>
<div className="mt-6 eyebrow">Clinic details</div>
<h1 className="page-title">{clinic.name}</h1>
<p className="page-lead">{clinic.address || 'Care team and consultation options'}</p>
<div className="mt-5 flex flex-wrap gap-2">
<span className="tag">
<Clock3 size={14} />{clinic.timezone}</span>
<span className="tag">
<MapPin size={14} />Active clinic</span>
</div>
<div className="mt-10 flex items-end justify-between">
<div>
<div className="eyebrow">Care team</div>
<h2 className="section-title">Doctors at this clinic</h2>
</div>
<span className="text-sm text-slate-500">{doctors.length} available</span>
</div>{doctors.length === 0 ? <div className="mt-4">
<EmptyState title="No doctors listed" message="This clinic has no active doctors available right now." />
</div> : <div className="mt-4 grid gap-4 md:grid-cols-2">{doctors.map((doctor) => <Link className="doctor-card" to={`/app/clinics/${clinic.id}/doctors/${doctor.id}`} key={doctor.id}>
<span className="doctor-avatar">
<Stethoscope size={20} />
</span>
<span>
<strong>{doctor.fullName}</strong>
<span>{doctor.specialization || 'General care'}</span>
</span>
<ArrowRight className="ml-auto text-slate-400" size={18} />
</Link>)}</div>}</div>;
}

export function DoctorPage() {
  const { clinicId, doctorId } = useParams(); const navigate = useNavigate(); const [doctor, setDoctor] = useState<PublicDoctor | null>(null); const [options, setOptions] = useState<{ consultationMode: string; durationMinutes: number; priceCents: number }[]>([]); const [error, setError] = useState<unknown>(null); const [loading, setLoading] = useState(true);
  useEffect(() => { if (!clinicId || !doctorId) return; void Promise.all([api.getDoctor(doctorId), api.getDoctorOptions(clinicId, doctorId)]).then(([doctorResult, optionsResult]) => { setDoctor(doctorResult.doctor); setOptions(optionsResult.options); }).catch(setError).finally(() => setLoading(false)); }, [clinicId, doctorId]);
  if (loading) return <LoadingState label="Loading doctor" />; if (error) return <ApiErrorState error={error} />; if (!doctor) return <EmptyState title="Doctor not found" message="This doctor is no longer available." />;
  const clinic = doctor.clinics.find((item) => item.id === clinicId);
  return <div>
<Link className="back-link" to={`/app/clinics/${clinicId}`}>
<ArrowLeft size={16} />Back to {clinic?.name || 'clinic'}</Link>
<div className="mt-7 flex flex-col gap-5 sm:flex-row sm:items-center">
<span className="doctor-avatar size-16">
<Stethoscope size={28} />
</span>
<div>
<div className="eyebrow">Doctor profile</div>
<h1 className="page-title mt-1">{doctor.fullName}</h1>
<p className="page-lead">{doctor.specialization || 'General care'}</p>
</div>
</div>
<section className="surface mt-8">
<div className="flex items-start justify-between gap-4">
<div>
<div className="eyebrow">Available consultations</div>
<h2 className="section-title">Choose how you want to meet</h2>
</div>
<HeartPulse className="text-teal-600" size={25} />
</div>{options.length === 0 ? <div className="mt-6">
<EmptyState title="No options configured" message="This doctor has not published consultation options yet." />
</div> : <div className="mt-6 grid gap-3 sm:grid-cols-2">{options.map((option) => <button className="option-card text-left" key={`${option.consultationMode}-${option.durationMinutes}`} onClick={() => navigate(`/app/clinics/${clinicId}/doctors/${doctorId}/book`, { state: option })}>
<span>
<strong>{option.consultationMode.replace('_', ' ')}</strong>
<span>{option.durationMinutes} minutes</span>
</span>
<span className="font-display text-lg font-semibold text-teal-800">{formatMoney(option.priceCents)}</span>
</button>)}</div>}</section>
</div>;
}
function formatMoney(cents: number) { return new Intl.NumberFormat('en-IN', { style: 'currency', currency: 'INR', maximumFractionDigits: 0 }).format(cents / 100); }
